# IOB Port Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a faithful port of `usim/iob.c`'s `WITH_SDL3` code path (the CADR I/O board), and correct `Keyboard.cs`/`Mouse.cs`/`WpfKeyTranslator.cs`, which turned out to be entirely invented relative to the real keyboard/mouse wire protocol IOB owns.

**Architecture:** Three tasks. Task 1 builds `Iob.cs` fully standalone (CSR, clocks, chaos-stub, gpio/serial/undoc-stub, beep callback, keyboard/mouse-ready methods) with its own tests — no wiring yet. Task 2 replaces `WpfKeyTranslator.TranslateKey`'s X11-keysym switch body with the real, independently-computed Cadet-scancode table — also fully standalone, no dependency on `Iob`. Task 3 wires everything together: `Keyboard.cs`/`Mouse.cs` are rewritten to own the real scancode/mouse-register encoding and call the wired `Iob`, `BusAdaptor`/`MachineControl` get real dispatch, and the entirely-invented `IOBus.cs` (plus its one caller, `MachineControl.RegisterIODevices()`) is deleted outright.

**Tech Stack:** C#, .NET 8.0, WPF. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-09-18-iob-design.md`

## Global Constraints

- No new NuGet dependencies.
- Every octal/hex literal in this plan's code was independently verified — see the spec's Octal Literals table (including the full hardcoded Cadet-scancode table, computed via Python from the real `.defs` files already in this repo, not transcribed by hand).
- Register width is 16 bits (`ushort`), not 32 — a genuine difference from every prior XBus-I/O device port (disk, TV, color TV).
- No `UsimState.*Enabled` gate on `Iob` — it is not optional hardware.
- The Cadet-scancode mapping table's *runtime, file-loading architecture* is not replicated (the real values are hardcoded in C# instead); the SDL3 keyboard thread's queue/backoff mechanism is not replicated; the keyboard-triggered cold/warm-reboot hotkey is dropped entirely (`MachineControl.PowerOn()` is already reachable through existing paths); real Chaosnet networking (`uch11.c`) stays a separate future port behind a minimal boot-safe stub.

// Iob.cs - Faithful port of usim/iob.c's WITH_SDL3 code path (the CADR I/O
// board). Real Chaosnet networking (usim/uch11.c) is a separate, future
// port; the four Chaos registers real boot activity touches get a minimal,
// safe idle stub here. The keyboard-triggered cold/warm-reboot hotkey is
// dropped entirely -- see docs/superpowers/specs/2026-09-18-iob-design.md.

using System;
using System.Diagnostics;

namespace Usim;

public class Iob
{
    private const uint ChaosMyAddrPlaceholder = 0x7C20; // 0177040 octal, uch11.c's LOCAL-CADR fallback

    private readonly UCode _ucode;

    public Action<int>? OnBeep { get; set; }

    private uint _csr;
    private uint _scancode;
    private uint _mouseX;
    private uint _mouseY;
    private ushort _the60CycleClock;
    private uint _intervalTimerTicksRemaining;
    private readonly Stopwatch _usecStopwatch = new();
    private bool _usecStarted;

    public Iob(UCode ucode)
    {
        _ucode = ucode;
    }

    private bool IsKeyboardInterruptEnabled() => (_csr & 0x4) != 0;
    private bool IsMouseInterruptEnabled() => (_csr & 0x2) != 0;

    /// <summary>
    /// Faithful port of iob_bus_reset (usim/iob.c:428-431), which has a
    /// genuinely empty body -- IOB state is never cleared by any reset in
    /// the real C. Kept as an explicit method purely for wiring symmetry
    /// with Tv/ColorTv.
    /// </summary>
    public void Reset()
    {
    }

    /// <summary>
    /// Faithful port of iob_set_keyboard_ready (usim/iob.c:88-107, WITH_SDL3
    /// branch), minus the cold/warm-boot sentinel check (that hotkey
    /// mechanism is dropped -- see the design's Decisions). Called directly
    /// from Keyboard's key-down path -- no queue, no retry/backoff.
    /// </summary>
    public void SetKeyboardReady(uint scancode)
    {
        _scancode = scancode;
        _csr |= 1u << 5;

        if (IsKeyboardInterruptEnabled())
        {
            _ucode.AssertUnibusInterrupt(0xB0);
        }
    }

    /// <summary>
    /// Faithful port of iob_set_mouse_ready (usim/iob.c:112-132, WITH_SDL3
    /// branch). rawX/rawY are always 0 -- WPF delivers discrete positions,
    /// not raw quadrature-encoder ticks, so there is no realistic value to
    /// synthesize (see the design's Decisions).
    /// </summary>
    public void SetMouseReady(int x, int y, bool tailPressed, bool middlePressed, bool headPressed)
    {
        _mouseX = (uint)(x & 0x7777);
        _mouseY = ((tailPressed ? 1u : 0u) << 12)
            | ((middlePressed ? 1u : 0u) << 13)
            | ((headPressed ? 1u : 0u) << 14)
            | (uint)(y & 0x7777);
        _csr |= 1u << 4;

        if (IsMouseInterruptEnabled())
        {
            _ucode.AssertUnibusInterrupt(0xB4);
        }
    }

    /// <summary>Faithful port of iob_unibus_read (usim/iob.c:138-303, WITH_SDL3 branch + shared registers).</summary>
    public ushort Read(uint offset)
    {
        switch (offset)
        {
            case 0x20: // kbd low
                {
                    ushort v = (ushort)(_scancode & 0xFFFF);
                    _csr &= ~(1u << 5);
                    return v;
                }

            case 0x22: // kbd high
                return (ushort)((_scancode >> 16) & 0xFFFF);

            case 0x24: // mouse y
                {
                    ushort v = (ushort)_mouseY;
                    _csr &= ~(1u << 4);
                    return v;
                }

            case 0x26: // mouse x
                return (ushort)_mouseX;

            case 0x28: // beep
                OnBeep?.Invoke(744); // usim/iob.c:209-211's own fixed default: 744us half-wavelength (~672Hz)
                return 0;

            case 0x2A: // csr
                return (ushort)_csr;

            case 0x30: // usec clock low
                return (ushort)(ElapsedMicroseconds() & 0xFFFF);

            case 0x32: // usec clock high
                return (ushort)((ElapsedMicroseconds() >> 16) & 0xFFFF);

            case 0x34: // 60Hz clock
                return _the60CycleClock;

            case 0x36: // gpio
                TraceLog.Instance.Warning(TraceCategory.IOBus, "iob: read from gpio");
                return 0;

            case 0x40: // ch_csr
                return 0;

            case 0x42: // ch_myaddr
                return (ushort)ChaosMyAddrPlaceholder;

            case 0x44: // ch_rbf
            case 0x46: // ch_rbc
                return 0;

            case 0x48: // ch_nop (obsolete)
            case 0x4C:
            case 0x4E:
                TraceLog.Instance.Warning(TraceCategory.IOBus, $"iob: chaos read obsolete unibus address: 0x{offset:X}");
                return 0;

            case 0x4A: // ch_xmt
                return (ushort)ChaosMyAddrPlaceholder;

            case 0x60: // serial
            case 0x62:
            case 0x64:
            case 0x66:
                TraceLog.Instance.Warning(TraceCategory.IOBus, $"iob: read from serial interface: 0x{offset:X}");
                return 0;

            case 0x68: // undocumented
            case 0x6A:
            case 0x6C:
            case 0x6E:
                TraceLog.Instance.Warning(TraceCategory.IOBus, $"iob: read from unknown device: 0x{offset:X}");
                return 0;

            default:
                TraceLog.Instance.Warning(TraceCategory.IOBus, $"iob: read from unknown offset: 0x{offset:X}");
                return 0;
        }
    }

    /// <summary>Faithful port of iob_unibus_write (usim/iob.c:307-384).</summary>
    public void Write(uint offset, ushort v)
    {
        switch (offset)
        {
            case 0x20: // kbd low
            case 0x22: // kbd high
            case 0x24: // mouse y
            case 0x26: // mouse x
                break; // real C: DEBUG trace only, no state change

            case 0x28: // beep
                OnBeep?.Invoke(v);
                break;

            case 0x2A: // csr
                _csr = (_csr & ~0xFu) | (v & 0xFu);
                break;

            case 0x30: // usec clock
            case 0x32:
                break; // real C: DEBUG trace only, no state change (read-only in practice)

            case 0x34: // interval timer set
                {
                    const double tickMicroseconds = 16000.0; // ~16ms per Tick(), matching WpfBackend/the headless loop
                    _intervalTimerTicksRemaining = (uint)Math.Ceiling(16.0 * v / tickMicroseconds);
                }
                break;

            case 0x40: // ch_csr
            case 0x42: // ch_myaddr write (buffer write in the real C)
            case 0x44: // ch_rbf
            case 0x46: // ch_rbc
            case 0x48: // ch_nop
            case 0x4A: // ch_xmt
                break; // accepted, ignored -- real Chaosnet networking is out of scope

            case 0x60: // serial
            case 0x62:
            case 0x64:
            case 0x66:
                TraceLog.Instance.Warning(TraceCategory.IOBus, $"iob: uart write: 0x{offset:X} v=0x{v:X}");
                break;

            default:
                TraceLog.Instance.Warning(TraceCategory.IOBus, $"iob: write to unknown offset: 0x{offset:X} v=0x{v:X}");
                break;
        }
    }

    /// <summary>
    /// Faithful port of run_at_60hz's the_60_cycle_clock++ (usim/usim.c) plus
    /// the tick-counted interval-timer approximation (see the design).
    /// </summary>
    public void Tick()
    {
        _the60CycleClock++;

        if (_intervalTimerTicksRemaining > 0)
        {
            _intervalTimerTicksRemaining--;
            if (_intervalTimerTicksRemaining == 0)
            {
                _ucode.AssertUnibusInterrupt(0xBC);
            }
        }
    }

    private long ElapsedMicroseconds()
    {
        if (!_usecStarted)
        {
            _usecStopwatch.Start();
            _usecStarted = true;
            return 0;
        }

        return _usecStopwatch.Elapsed.Ticks / 10; // Stopwatch ticks are 100ns
    }
}

        ucode = new UCode(new MainMemory());
        return new Iob(ucode);
    }

    private static bool TestCsrWriteMasksToLow4Bits()
    {
        Console.WriteLine("Test: CSR write only touches the low 4 bits");
        try
        {
            var iob = MakeIob(out _);

            iob.Write(0x2A, 0xFFFF);
            Assert(iob.Read(0x2A) == 0xF, $"CSR write masks to low 4 bits, got 0x{iob.Read(0x2A):X}");

            Console.WriteLine("  CSR-write-mask test passed\n");
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"  CSR-write-mask test failed: {ex.Message}\n");
            return false;
        }
    }

    private static bool TestKeyboardReadySetAndClearOnLowWordOnly()
    {
        Console.WriteLine("Test: SetKeyboardReady sets CSR bit 5; reading the low word clears it, reading the high word does not");
        try
        {
            var iob = MakeIob(out _);

            iob.SetKeyboardReady(0x00120034); // arbitrary scancode with distinct low/high halves
            Assert((iob.Read(0x2A) & 0x20) == 0x20, "keyboard-ready bit is set after SetKeyboardReady");

            ushort high = iob.Read(0x22); // high word first -- must not clear the ready bit
            Assert(high == 0x0012, $"kbd high returns the upper 16 bits, got 0x{high:X}");
            Assert((iob.Read(0x2A) & 0x20) == 0x20, "keyboard-ready bit is still set after reading kbd high");

            ushort low = iob.Read(0x20); // low word -- clears the ready bit
            Assert(low == 0x0034, $"kbd low returns the lower 16 bits, got 0x{low:X}");
            Assert((iob.Read(0x2A) & 0x20) == 0, "keyboard-ready bit is cleared after reading kbd low");

            Console.WriteLine("  Keyboard-ready set/clear-on-low-word-only test passed\n");

        catch (Exception ex)
        {
            Console.WriteLine($"  Keyboard-ready set/clear-on-low-word-only test failed: {ex.Message}\n");
            return false;
        }
    }

    {
        Console.WriteLine("Test: SetKeyboardReady asserts the Unibus interrupt only when CSR bit 2 (keyboard-interrupt-enable) is set");
        try
        {
            var iob = MakeIob(out UCode ucode);

            iob.SetKeyboardReady(0x1);
            Assert(!ucode.InterruptPendingFlag, "no interrupt when keyboard-interrupt-enable is clear");

            iob.Write(0x2A, 0x4); // interrupt-enable bit 2
            iob.SetKeyboardReady(0x2);
            Assert(ucode.InterruptPendingFlag, "interrupt asserted once keyboard-interrupt-enable is set");

            Console.WriteLine("  Keyboard-interrupt-gating test passed\n");

            return false;

            var iob = MakeIob(out _);

            iob.SetMouseReady(10, 20, false, false, false);
            Assert((iob.Read(0x2A) & 0x10) == 0x10, "mouse-ready bit is set after SetMouseReady");

            var iob = MakeIob(out UCode ucode);

            iob.SetMouseReady(1, 1, false, false, false);
            Assert(!ucode.InterruptPendingFlag, "no interrupt when mouse-interrupt-enable is clear");

            iob.SetMouseReady(2, 2, false, false, false);

            Console.WriteLine($"  Mouse-interrupt-gating test failed: {ex.Message}\n");
            return false;
        }

    private static bool TestMouseBitLayoutWithRawAlwaysZero()
    {
        Console.WriteLine("Test: mouse X/Y register bit layout matches the real 0x7777 mask, with rawX/rawY always 0, and tail/middle/head packed into Y at bits 12/13/14");
        try
        {
            var iob = MakeIob(out _);

            // x=0xFFFF exercises the mask on its own: every bit set, masking
            // must clear every 4th bit, leaving exactly 0x7777.
            int x = 0xFFFF;
            // y=0 is deliberate, not arbitrary: 0x7777's own top nibble is
            // 0111, meaning bits 12/13/14 are ALREADY set by the mask
            // whenever y's own top nibble has any bits set (e.g. y=0xFFFF
            // masks to 0x7777, which already has bits 12 and 14 set --
            // independently verified via Python before writing this test).
            // That would make tailPressed/headPressed's OR'd-in bits
            // indistinguishable from the mask's own contribution -- a
            // vacuous test. y=0 makes the masked contribution to those bits
            // exactly 0, so the toggle bits below are the only source of
            // any 1s there, making this a genuinely discriminating check.
            int y = 0;
            iob.SetMouseReady(x, y, tailPressed: true, middlePressed: false, headPressed: true);

            ushort mx = iob.Read(0x26);
            Assert(mx == 0x7777, $"mouse X is x masked by 0x7777 with rawX/rawY always 0, got 0x{mx:X}");

            ushort my = iob.Read(0x24);
            // bits 12=tail(1), 13=middle(0), 14=head(1) over a masked y of 0 -> 0x5000 exactly.
            ushort expectedY = (ushort)((1 << 12) | (0 << 13) | (1 << 14) | (y & 0x7777));
            Assert(my == expectedY, $"mouse Y packs tail/middle/head at bits 12/13/14 over the masked y, expected 0x{expectedY:X} got 0x{my:X}");

            Console.WriteLine("  Mouse-bit-layout test passed\n");
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"  Mouse-bit-layout test failed: {ex.Message}\n");
            return false;
        }
    }

    private static bool TestUsecClockStartsAtZeroThenIncreases()
    {
        Console.WriteLine("Test: the usec clock returns 0 on its first read, then a monotonically increasing value");
        try
        {
            var iob = MakeIob(out _);

            ushort firstLow = iob.Read(0x30);
            ushort firstHigh = iob.Read(0x32);

        {
            Console.WriteLine($"  Usec-clock test failed: {ex.Message}\n");

        }
    }

        }
        catch (Exception ex)
        {
            Console.WriteLine($"  60Hz-clock-increments-per-tick test failed: {ex.Message}\n");
            return false;
        }
    }

    private static bool TestIntervalTimerFiresAtComputedTickCount()
    {
        Console.WriteLine("Test: writing the interval timer register fires the Unibus interrupt at exactly the computed tick count");
        try
        {
            var iob = MakeIob(out UCode ucode);

            // 16 * v / 16000.0 ticks, ceiling. v=2000 -> 16*2000/16000 = 2.0 -> 2 ticks exactly.
            iob.Write(0x34, 2000);

            iob.Tick();
            Assert(!ucode.InterruptPendingFlag, "no interrupt after the first of two required ticks");

            iob.Tick();
            Assert(ucode.InterruptPendingFlag, "interrupt fires on exactly the second tick");

            Console.WriteLine("  Interval-timer-fires-at-computed-tick-count test passed\n");
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"  Interval-timer-fires-at-computed-tick-count test failed: {ex.Message}\n");
            return false;
        }
    }

    private static bool TestBeepCallbackOnReadAndWrite()
    {
        Console.WriteLine("Test: reading the beep register invokes OnBeep with the real fixed default (744); writing invokes it with the written value");
        try
        {
            var iob = MakeIob(out _);
            int? lastBeep = null;
            iob.OnBeep = v => lastBeep = v;

            iob.Read(0x28);
            Assert(lastBeep == 744, $"read-triggered beep uses the real fixed default 744, got {lastBeep}");

            lastBeep = null;
            iob.Write(0x28, 500);
            Assert(lastBeep == 500, $"write-triggered beep passes the written value through, got {lastBeep}");

            Console.WriteLine("  Beep-callback test passed\n");
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"  Beep-callback test failed: {ex.Message}\n");
            return false;
        }
    }

    private static bool TestGpioSerialUndocumentedStubsDoNotThrow()
    {
        Console.WriteLine("Test: GPIO/serial/undocumented registers return 0 and never throw, on read or write");
        try
        {
            var iob = MakeIob(out _);

            Assert(iob.Read(0x36) == 0, "gpio read returns 0");
            Assert(iob.Read(0x60) == 0, "serial read returns 0");
            Assert(iob.Read(0x68) == 0, "undocumented read returns 0");
            iob.Write(0x60, 0x1234); // must not throw

            Console.WriteLine("  GPIO/serial/undocumented-stubs test passed\n");
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"  GPIO/serial/undocumented-stubs test failed: {ex.Message}\n");
            return false;
        }
    }

    private static bool TestChaosStubRegistersDoNotThrow()
    {
        Console.WriteLine("Test: Chaos-interface stub registers return their documented idle values and never throw, on read or write");
        try
        {
            var iob = MakeIob(out _);

            Assert(iob.Read(0x40) == 0, "ch_csr reads 0 (idle)");
            Assert(iob.Read(0x42) == 0x7C20, $"ch_myaddr reads the placeholder address, got 0x{iob.Read(0x42):X}");
            Assert(iob.Read(0x44) == 0, "ch_rbf reads 0");
            Assert(iob.Read(0x46) == 0, "ch_rbc reads 0");
            Assert(iob.Read(0x4A) == 0x7C20, "ch_xmt echoes the placeholder address");
            iob.Write(0x40, 0xFFFF); // must not throw, must not change ch_csr's idle read
            Assert(iob.Read(0x40) == 0, "ch_csr write is accepted and ignored");

            Console.WriteLine("  Chaos-stub-registers test passed\n");
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"  Chaos-stub-registers test failed: {ex.Message}\n");
            return false;
        }
    }

    private static void Assert(bool condition, string message)
    {
        if (!condition)
            throw new Exception($"Assertion failed: {message}");
    }
}

            if ((Buttons & button) != (_lastButtons & button))
            {
                ClickCount++;
                ButtonPressed?.Invoke(button);
            }
        }
        else
        {
            Buttons &= ~button;
            if ((Buttons & button) != (_lastButtons & button))
            {
                ButtonReleased?.Invoke(button);
            }
        }

        switch (button)
        {
            case MouseButtons.Left:
                _tailToggle = !_tailToggle;
                break;
            case MouseButtons.Middle:
                _middleToggle = !_middleToggle;
                break;
            case MouseButtons.Right:
                _headToggle = !_headToggle;
                break;
        }

        ReportToIob();
    }

    public bool IsButtonPressed(MouseButtons button)
    {
        return (Buttons & button) != 0;
    }

    public (int dx, int dy) GetDelta()
    {
        int dx = X - _lastX;
        int dy = Y - _lastY;
        _lastX = X;
        _lastY = Y;
        return (dx, dy);
    }

    public void UpdatePosition(int x, int y)
    {
        MoveTo(x, y);
        TraceLog.Instance.Trace(TraceCategory.Mouse, TraceLevel.Verbose,
            $"Mouse moved to ({x}, {y})");
    }

    public void ButtonDown(MouseButtons button)
    {
        SetButton(button, true);
        TraceLog.Instance.Trace(TraceCategory.Mouse, TraceLevel.Debug,
            $"Mouse button down: {button}");
    }

    public void ButtonUp(MouseButtons button)
    {
        SetButton(button, false);
        TraceLog.Instance.Trace(TraceCategory.Mouse, TraceLevel.Debug,
            $"Mouse button up: {button}");
    }

    private void ReportToIob()
    {
        _iob?.SetMouseReady(X, Y, _tailToggle, _middleToggle, _headToggle);
    }
